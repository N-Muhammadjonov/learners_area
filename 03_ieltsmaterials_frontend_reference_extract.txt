/* --- One header, one size ------------------------------------------------
 *
 * Everything above styles the header. This section fixes its *geometry*, and
 * it exists because ten stylesheets each shipped their own copy of that
 * geometry and the copies disagreed. Measured, at 375px wide:
 *
 * reading, listening, mock, mock-results .top-left row, logo 52x52 round
 * index, login, account, books, speaking, writing
 * .top-left column, logo 88px auto
 *
 * The second group stacks the logo above the brand text, so the same header
 * is one row on a listening page and two rows on the homepage - which is the
 * "different size on every page" this fixes. The breakpoint disagreed too:
 * mock compacts at 720px, most pages at 760px, mock-results at 840px, so
 * between 760 and 840 three pages were compact and seven were not.
 *
 * The rules below are scoped to `.topbar` (0,2,1) where the page copies are
 * bare element/class selectors (0,1,1). Specificity, not source order,
 * because source order is exactly what was already unreliable: this sheet is
 * linked after the page's own on Django-rendered pages but the page rules sit
 * in breakpoints this sheet does not share, so at 800px wide a `.brand img`
 * rule from mock-results.css applied with nothing here to answer it.
 *
 * 840px is the widest of the three existing breakpoints on purpose: every
 * page keeps compacting no later than it already did, and now they all
 * compact at the same width.
 */

/* One scroller per page.
 *
 * Every page sheet extracted from the legacy site carries this pair:
 *
 * html { overflow-x: hidden; }
 * body { ...; overflow-x: hidden; ... }
 *
 * and it is the pair, not either line, that is the bug.
 *
 * Two rules meet here. CSS Overflow 3 §3.3: when one axis is not `visible`,
 * the other's `visible` computes to `auto`  so `overflow-x: hidden` means
 * `overflow-y: auto` as well, and the element becomes a scroll container.
 * §3.5: the viewport's own overflow is propagated from the root element, and
 * only if the root is `visible` is it taken from <body> instead. So:
 *
 * html visible + body hidden body's value goes to the viewport,
 * body itself is visible. One scroller.
 * (This is /aslmock, measured: 0px.)
 * html hidden + body hidden html's value goes to the viewport, and
 * body's applies to body. Two scrollers.
 *
 * The second is what these pages ship. Measured on /listening, signed in,
 * 122 cards, before this file existed:
 *
 * scroll chain under the cursor BODY -> HTML
 * body's own scrollbar 15px wide, beside the window's
 * body can travel 68px
 * html can travel 21,516px
 *
 * Two scrollbars on the right edge, which is what it looks like, and a wheel
 * gesture split between them, which is what it feels like: one tick down from
 * the top moved body 0 -> 68 and only then html 0 -> 125, so the page does not
 * begin to move until the inner scroller is used up, and on the way back up
 * the last 68px of the gesture go into the inner one instead of the page. With
 * this file loaded the same tick moves html 0 -> 225 and body not at all.
 *
 * What body was scrolling is decoration: .shell::after is
 * position: absolute; bottom: -66px, so it hangs below the shell and into
 * body's scrollport. The homepage traps 2px and /reading 8px  the size does
 * not matter, its existence does.
 *
 * `html { overflow-x: hidden }` is restated rather than assumed, because
 * login.css, mock.css and aslmock.css put the rule only on <body>: without it
 * here, `body { overflow-x: visible }` would leave those pages with nothing
 * clipping sideways. On the root it clips the same things and creates no inner
 * scroller  /listening still hides 84px of horizontal decoration
 * (.shell::before is right: -84px) with no horizontal scrollbar.
 *
 * Deliberately a separate file, like listing-perf.css: reading.css,
 * listening.css, writing.css and the rest are output from
 * scripts/extract_page_assets.py, and anything written into them is lost the
 * next time it runs.
 *
 * Specificity is left alone on purpose. mock.css's `body.mock-frame-open` and
 * login.css's `body.login-blocked-lock` set `overflow: hidden` to lock the
 * page behind an overlay; a class beats the bare element selector below, so
 * those still win.
 *
 * Not loaded by content/books.html and content/speaking.html, which carry the
 * same pair inline and are served from CONTENT_ROOT rather than rendered.
 * They are fixed in place.
 */

html {
 overflow-x: hidden;
}

body {
 overflow-x: visible;
}

/* And then the one scrollbar is not drawn.
 *
 * Asked for: the 15px grey trough with its two arrow buttons is Windows
 * furniture on a page that has none, and it sat against the cards. The
 * element stays a scroll container  this hides the indicator, it does not
 * take the scrolling away. Measured after the change: the wheel still moves
 * the page, and 38,140px of travel is still there to move through.
 * Keyboard scrolling is not measured here  PageDown does not scroll under
 * the automation harness with or without these rules, so there was no control
 * case to compare against. Nothing in `scrollbar-width` or the pseudo-element
 * touches key handling, but that is a reading of the spec, not a measurement.
 *
 * The trade is the indicator itself: on a 122-card listing nothing now shows
 * how far down the page you are, and there is no thumb to drag.
 *
 * Nothing changes on macOS or iOS, where scrollbars are already overlaid and
 * invisible until you scroll. This is for Windows and Linux.
 *
 * It also makes the width consistent rather than less so. A reserved
 * scrollbar made every scrolling page 15px narrower than one that fit, so the
 * layout shifted sideways between /account and /listening. With no scrollbar
 * reserved anywhere, every page is the full width.
 *
 * Three declarations because three engines have three answers, and the
 * property is on the scroll container  which is <html>, since the root's
 * overflow is what propagates to the viewport (the rule above). `body` is
 * named too: the pseudo-element is matched against whichever element the
 * viewport's scrollbar is taken from, and that is not the same choice in
 * every version.
 */
html,
body {
 scrollbar-width: none; /* Firefox 64+ */
 -ms-overflow-style: none; /* Edge 12-18, IE 10+ */
}

html::-webkit-scrollbar,
body::-webkit-scrollbar {
 /* Chrome, Edge, Safari, and every Chromium-based browser on Windows. */
 width: 0;
 height: 0;
 display: none;
}

/* Make the server the source of truth for "who is signed in".
 *
 * Every shipped script  progress-tracker.js, header-enhance.js, account.js,
 * ai-recommendation.js  answers that question by reading
 * localStorage['ielts_user_email'], which only login.js ever writes. The
 * signed session cookie, which is what the server actually trusts, never
 * reaches them.
 *
 * The two drift apart in ordinary use:
 *
 * - Clearing site data leaves the cookie intact but the key gone, so the
 * page decides nobody is signed in and bounces a valid session to /login.
 * - Signing in as a second person on a shared browser leaves the first
 * email behind if the sign-in did not complete, so the header names an
 * account the visitor is not using. Observed with a deleted account still
 * showing in the header after its row was gone.
 *
 * Rewriting four scripts to read a new global would be the larger change.
 * This writes the server's answer into the key they already read, before they
 * run, and fires the `storage`-style refresh header-enhance.js listens for.
 *
 * It is display only. Nothing here grants access: every endpoint identifies
 * the caller from the session cookie and treats a supplied email as a
 * cross-check (TALABLAR §2).
 */
(function () {
 var KEY = 'ielts_user_email';
 var body = document.body;
 if (!body) return;

 // Absent attribute means the server did not render an opinion; leave
 // whatever is stored alone rather than signing the visitor out.
 if (!body.hasAttribute('data-session-email')) return;
 var email = (body.getAttribute('data-session-email') || '').trim().toLowerCase();

 var stored;
 try {
 stored = (localStorage.getItem(KEY) || '').trim().toLowerCase();
 } catch (e) {
 return;
 }
 if (stored === email) return;

 try {
 if (email) {
 localStorage.setItem(KEY, email);
 } else {
 localStorage.removeItem(KEY);
 }
 } catch (e) {
 return;
 }

 // header-enhance.js repaints on the storage event, which does not fire in
 // the tab that made the change.
 try {
 window.dispatchEvent(new StorageEvent('storage', {
 key: KEY, newValue: email || null, storageArea: localStorage,
 }));
 } catch (e) {}
})();

 function bootstrapCacheKey(email) {
 return BOOTSTRAP_CACHE_PREFIX + String(email || 'guest').toLowerCase();
 }

 function loadBootstrapCache(email) {
 if (!email) return null;
 try {
 var raw = localStorage.getItem(bootstrapCacheKey(email));
 if (!raw) return null;
 var parsed = JSON.parse(raw);
 if (!parsed || typeof parsed.savedAt !== 'number') return null;
 var age = Date.now() - parsed.savedAt;
 if (age < 0 || age > BOOTSTRAP_CACHE_TTL_MS) return null;
 return parsed;
 } catch (e) {
 return null;
 }
 }

 function saveBootstrapCache(email, payload) {
 if (!email || !payload) return;
 try {
 localStorage.setItem(
 bootstrapCacheKey(email),
 JSON.stringify({ savedAt: Date.now(), payload: payload })
 );
 } catch (e) {}
 }

 // Hydrate module-level state from localStorage if a fresh snapshot exists.
 // Called once on script load so the page can render with real premium/material
 // info immediately, without waiting for any network round-trip.
 function hydrateFromCache() {
 var email = getEmail();
 if (!email) return false;
 var entry = loadBootstrapCache(email);
 if (!entry || !entry.payload) return false;
 applyPremiumPayload(entry.payload.premium, email);
 applyMaterialsPayload(entry.payload.materials);
 return true;
 }

 // Single-call alternative to running initSync + initPremiumAccess +
 // initMaterialAccess separately. Hits /api/bootstrap once and seeds all
 // three module-level state variables from one response, so a page load
 // costs 1 round trip instead of 3.
 async function initBootstrap(forceRefresh) {
 if (forceRefresh) {
 bootstrapPromise = null;
 bootstrapAttempted = false;
 }
 if (bootstrapPromise) return bootstrapPromise;
 var email = getEmail();
 if (!email) return null;

 // If we already revalidated very recently, skip another network call.
 if (!forceRefresh && bootstrapAttempted) {
 var entry = loadBootstrapCache(email);
 if (entry && (Date.now() - entry.savedAt) < BOOTSTRAP_REVALIDATE_AFTER_MS) {
 return null;
 }
 }

 bootstrapAttempted = true;
 bootstrapPromise = (async function () {
 var res;
 try {
 res = await fetch(apiUrl('/api/bootstrap?email=' + encodeURIComponent(email)), {
 method: 'GET',
 headers: authHeaders({ 'Accept': 'application/json', 'Cache-Control': 'no-store' }),
 cache: 'no-store'
 });
 } catch (err) {
 return null;
 }
 var text = await res.text();
 var payload = {};
 if (text) {
 try { payload = JSON.parse(text); } catch (e) {}
 }
 if (!res.ok) {
 if (isBlockedDeviceMessage(payload && payload.error) && !blockedNoticeShown) {
 showBlockedAccess(payload.error);
 }
 return null;
 }

 // Seed account/progress (same effect as initSync resolving)
 try {
 var migrated = migrateGuestProgressToAccount(email);
 var local = migrated || loadLocalData();
 var remote = normalizeData((payload.account && payload.account.progress) || {});
 var merged = mergeData(local, remote);
 saveLocalData(merged);
 pushProgressToServer(merged);
 syncResultCached = merged;
 } catch (e) {}

 applyPremiumPayload(payload.premium, email);
 applyMaterialsPayload(payload.materials);

 // Persist a snapshot so the next page open can render from local
 // storage instantly without waiting on the network.
 saveBootstrapCache(email, {
 premium: payload.premium || {},
 materials: payload.materials || { items: [] }
 });

 dispatchProgressReady('fresh');

 return {
 progress: syncResultCached,
 premium: premiumAccessState,
 materials: materialAccessState
 };
 })().finally(function () {
 bootstrapPromise = null;
 });
 return bootstrapPromise;
 }

 // to offer a login before the test opens, and that offer is the only
 // thing standing between a signed-out visitor and a premium test for as

 materialAccessState = { loaded: true, items: [], byId: {}, byPath: {} };

 materialAccessPromise = fetchMaterialAccess()
 .then(function (payload) {

 var byId = {};
 var byPath = {};
 items.forEach(function (item) {
 var href = String((item && item.href) || '');
 var materialId = String((item && item.material_id) || deriveMaterialIdFromHref(href));
 var pathKey = normalizeMaterialPath(href);
 var normalizedItem = {
 materialId: materialId,
 href: href,
 title: String((item && item.title) || ''),
 section: String((item && item.section) || ''),
 isPremium: !!(item && item.is_premium),
 audience: String((item && item.audience) || ((item && item.is_premium) ? 'premium' : 'all')).toLowerCase(),

 updatedBy: String((item && item.updated_by) || '')

 byId[materialId] = normalizedItem;

 materialAccessState = {
 loaded: true,
 items: items,

 return materialAccessState;
 })
 .catch(function () {
 materialAccessState = {

 function showBlockedAccess(message) {
 blockedNoticeShown = true;
 // TALABLAR §10: a device blocked inside the mock iframe has to move the
 // warning to the top browser window. The modal below is persistent - no
 // close button, no actions - so left in the frame it is a locked panel
 // inside a running mock: the candidate cannot dismiss it, cannot read why,
 // and the mock chrome around it still looks like a test in progress.
 //
 // The server covers the other half of the same rule: a blocked page
 // *navigation* is redirected to /account?access=blocked-device, and

 var framed = topWindowIfFramed();
 if (framed) {
 try {
 framed.location.replace(getBlockedDeviceAccountUrl());
 return;
 } catch (e) {}
 }
 openPremiumModal(
 message || 'You are banned because this premium account was opened from a fourth device.',

 );
 }

 function normalizeData(parsed) {
 var fresh = baseData();
 var data = Object.assign({}, fresh, parsed || {});
 var parsedTests = (parsed && parsed.tests) || {};

 Object.keys(fresh.tests).forEach(function (section) {
 data.tests[section] = Object.assign({}, fresh.tests[section], parsedTests[section] || {});
 if (!Array.isArray(data.tests[section].results)) data.tests[section].results = [];
 });

 data.books = Object.assign({}, fresh.books, (parsed && parsed.books) || {});
 data.books.used = Object.assign({}, fresh.books.used, data.books.used || {});
 data.drafts = Object.assign({}, fresh.drafts, (parsed && parsed.drafts) || {});
 if (!Array.isArray(data.logins)) data.logins = [];
 if (!Array.isArray(data.activity)) data.activity = [];
 return data;
 }

 function mergeDrafts(localDrafts, remoteDrafts) {
 var merged = {};
 var keys = Object.keys(remoteDrafts || {}).concat(Object.keys(localDrafts || {}));
 keys.forEach(function (key) {
 var local = (localDrafts || {})[key] || {};
 var remote = (remoteDrafts || {})[key] || {};
 var localUpdated = String(local.updated_at || '');
 var remoteUpdated = String(remote.updated_at || '');
 merged[key] = remoteUpdated > localUpdated ? remote : local;
 });
 return merged;
 }

 function mergeUsedBooks(localUsed, remoteUsed) {
 var merged = {};
 [remoteUsed || {}, localUsed || {}].forEach(function (source) {
 Object.keys(source).forEach(function (key) {
 var incoming = source[key] || {};
 var existing = merged[key] || { label: incoming.label || key, count: 0, last_used: '' };
 merged[key] = {
 label: incoming.label || existing.label || key,
 count: Math.max(Number(existing.count) || 0, Number(incoming.count) || 0),
 last_used: (existing.last_used || '') > (incoming.last_used || '') ? (existing.last_used || '') : (incoming.last_used || '')
 };
 });
 });
 return merged;
 }

 function mergeResults(localResults, remoteResults) {
 var seen = {};
 var merged = [];
 (remoteResults || []).concat(localResults || []).forEach(function (entry) {
 if (!entry) return;
 var key = entry.attempt_id || [entry.label || '', entry.score || 0, entry.total || 0, entry.submitted_at || ''].join('|');
 if (seen[key]) return;
 seen[key] = true;
 merged.push(entry);
 });
 merged.sort(function (a, b) {
 return String(b.submitted_at || '').localeCompare(String(a.submitted_at || ''));
 });
 return merged.slice(0, MAX_RESULTS_PER_SECTION);
 }

 merged.push(entry);

 merged.sort(function (a, b) {

 function mergeData(localData, remoteData) {
 var local = normalizeData(localData);
 var remote = normalizeData(remoteData);
 var merged = baseData();

 Object.keys(merged.tests).forEach(function (section) {
 var lt = local.tests[section] || {};
 var rt = remote.tests[section] || {};
 var results = mergeResults(lt.results, rt.results);
 var latestEntry = results[0] || null;
 var bestScore = null;
 results.forEach(function (entry) {
 var score = Number(entry && entry.score);
 if (isFinite(score)) bestScore = bestScore == null ? score : Math.max(bestScore, score);
 });
 merged.tests[section] = {
 visits: Math.max(Number(lt.visits) || 0, Number(rt.visits) || 0),
 opens: Math.max(Number(lt.opens) || 0, Number(rt.opens) || 0),
 last_opened: String(lt.last_opened || '') > String(rt.last_opened || '') ? String(lt.last_opened || '') : String(rt.last_opened || ''),
 latest_score: latestEntry ? latestEntry.score : (rt.latest_score != null ? rt.latest_score : lt.latest_score),
 best_score: bestScore != null ? bestScore : (rt.best_score != null ? rt.best_score : lt.best_score),
 last_submitted: latestEntry ? latestEntry.submitted_at : (String(lt.last_submitted || '') > String(rt.last_submitted || '') ? String(lt.last_submitted || '') : String(rt.last_submitted || '')),
 results: results
 };
 });

 merged.books = {
 visits: Math.max(Number(local.books.visits) || 0, Number(remote.books.visits) || 0),
 total_used: Math.max(Number(local.books.total_used) || 0, Number(remote.books.total_used) || 0),
 used: mergeUsedBooks(local.books.used, remote.books.used)
 };
 merged.drafts = mergeDrafts(local.drafts, remote.drafts);
 merged.logins = mergeLogins(local.logins, remote.logins);
 merged.activity = mergeActivity(local.activity, remote.activity);
 return normalizeData(merged);
 }

 async function initSync(forceRefresh) {
 var email = getEmail();
 if (!email) return baseData();
 // Reuse the merged data from the first sync so a second caller within
 // the same page load doesn't re-fetch /api/account-data.
 if (syncResultCached && !forceRefresh) return syncResultCached;
 if (syncPromise) return syncPromise;

 var migrated = migrateGuestProgressToAccount(email);
 var local = migrated || loadLocalData();
 syncPromise = fetchAccountData(email)
 .then(function (payload) {
 var remote = normalizeData(payload.progress || {});
 var merged = mergeData(local, remote);
 saveLocalData(merged);
 pushProgressToServer(merged);
 syncResultCached = merged;
 return merged;
 })
 .catch(function (error) {
 if (isBlockedDeviceMessage(error && error.message) && !blockedNoticeShown) {
 showBlockedAccess(error.message);
 }
 pushProgressToServer(local);
 syncResultCached = local;
 return local;
 })
 .finally(function () {
 syncPromise = null;
 });

 return syncPromise;
 }

 // Ask the server to mark an attempt, and bring back the key for review.
 //
 // The page calls this instead of marking itself. Two requests: the first
 // records the attempt and gets the score, the second returns the answer key
 //  which the server only releases once an attempt exists, so the order
 // matters and is not an optimisation to remove.
 //
 // Returns null rather than throwing when the server cannot be reached, so a
 // page can fall back to whatever it still holds. While the keys are still in
 // the pages that fallback is exact; once they are gone it is the difference
 // between a score and an apology, which is why this has to be reliable
 // before they go.
 // What a page does with the server's answer, in one place.
 //
 // Returns true when the paper can be marked and the page should show its
 // results, false when it cannot and has been told so instead.
 //
 // The false case is the one that matters. Once the answer keys leave the
 // pages, a failed request means the page has nothing to mark with  and
 // measured before this existed, it went ahead anyway: three correct answers
 // were counted as zero, no message, and the attempt was lost. A student
 // reading that has no way to know it was wrong.
 function applyMarking(marked, correctAnswers) {
 if (marked && marked.answers) {
 Object.assign(correctAnswers, marked.answers);
 }

 var usable = correctAnswers && Object.keys(correctAnswers).some(function (key) {
 var value = correctAnswers[key];
 return Array.isArray(value) ? value.length > 0 : Boolean(value);
 });
 if (usable) return true;

 showMarkingUnavailable();
 return false;
 }

 function markAttempt(href, submittedAnswers) {
 var path = String(href || window.location.pathname).split('?')[0].split('#')[0];
 var marked = null;

 return fetchWithCsrf(apiUrl('/api/attempt'), {
 method: 'POST',
 headers: authHeaders({ 'Content-Type': 'application/json', 'Accept': 'application/json' }),
 cache: 'no-store',
 body: JSON.stringify({ href: path, answers: submittedAnswers || {} })
 }).then(function (response) {
 if (!response.ok) return null;
 return response.json();
 }).then(function (payload) {
 if (!payload) return null;
 marked = payload;
 serverMarkedPaths[path] = true;
 return fetch(apiUrl('/api/attempt/answers?') + new URLSearchParams({ href: path }), {
 headers: authHeaders({ 'Accept': 'application/json' }),
 cache: 'no-store'
 });
 }).then(function (response) {
 if (!response || !response.ok) return marked;
 return response.json().then(function (key) {
 marked.answers = (key && key.answers) || {};
 return marked;
 });
 }).catch(function () {
 return marked;
 });
 }

 //
 // Keyed by path, not by attempt_id. Keying on the id looked right and was

 // trackTestResult makes up its own, so the two never matched and one
 // sitting of a listening full test was recorded as two attempts.
 var serverMarkedPaths = {};

 function submitAttemptForMarking(section, entry, submittedAnswers) {
 if (!submittedAnswers || typeof submittedAnswers !== 'object') return;
 if (!getEmail()) return;
 // The page already went through markAttempt(); submitting again would
 // record a second attempt for one sitting.
 if (serverMarkedPaths[currentMarkedPath()]) return;

 var href = String(entry && entry.href || window.location.pathname);
 // Strip the query the runner adds  ?redo=1, ?mock=&  so the server sees
 // the material and not the way it was opened.
 href = href.split('?')[0].split('#')[0];

 fetchWithCsrf(apiUrl('/api/attempt'), {
 method: 'POST',
 headers: authHeaders({ 'Content-Type': 'application/json', 'Accept': 'application/json' }),
 cache: 'no-store',
 body: JSON.stringify({ href: href, answers: submittedAnswers })
 }).then(function (response) {
 if (!response.ok) return null;
 return response.json();
 }).then(function (marked) {
 if (!marked || typeof marked.score !== 'number') return;
 if (marked.score === entry.score && marked.total === entry.total) return;

 // They disagreed. The server is the record, so replace ours rather than
 // leaving two numbers for the same attempt.
 var data = loadData();
 var bucket = data.tests[section];
 if (!bucket) return;
 bucket.results = (bucket.results || []).map(function (item) {
 return item && item.attempt_id === entry.attempt_id
 ? Object.assign({}, item, { score: marked.score, total: marked.total })
 : item;
 });
 bucket.latest_score = marked.score;
 bucket.best_score = bucket.best_score == null
 ? marked.score
 : Math.max(bucket.best_score, marked.score);
 saveData(data);
 }).catch(function () {
 // Offline, or the test is one a person marks. The local record stands.
 });
 }

 function trackTestResult(section, details) {
 if (!section) return;
 if (isSavedAttemptReview() && !(details && details.allow_review_tracking)) return;
 var data = loadData();
 if (!data.tests[section]) return;

 var score = Number(details && details.score);
 var total = Number(details && details.total);
 if (!isFinite(score) || !isFinite(total) || total <= 0) return;

 var label = (details && details.label) || (section.charAt(0).toUpperCase() + section.slice(1) + ' test');
 var submittedAt = nowIso();
 var attemptId = (details && details.attempt_id) || buildAttemptId(section, submittedAt);
 var answersSnapshot = details && details.answers;
 if (answersSnapshot && typeof answersSnapshot === 'object') {
 data.drafts = data.drafts || {};
 data.drafts[attemptId] = {
 answers: answersSnapshot,
 updated_at: submittedAt
 };
 }
 var entry = {
 attempt_id: attemptId,
 test_id: (details && details.test_id) || '',
 href: (details && details.href) || (window.location.pathname + window.location.search + window.location.hash),
 label: label,
 score: score,
 total: total,
 submitted_at: submittedAt
 };

 data.tests[section].results.unshift(entry);
 if (data.tests[section].results.length > MAX_RESULTS_PER_SECTION) {
 data.tests[section].results = data.tests[section].results.slice(0, MAX_RESULTS_PER_SECTION);
 }
 data.tests[section].latest_score = score;
 data.tests[section].last_submitted = submittedAt;
 data.tests[section].best_score = data.tests[section].best_score == null
 ? score
 : Math.max(data.tests[section].best_score, score);

 addActivity(data, 'Submitted ' + label + ': ' + score + '/' + total);
 saveData(data);

 // Have the server mark it too, and take its answer over ours.
 //
 // The score above is worked out here, from an answer key this page holds.
 // That key is being moved into the database so the page stops holding it,
 // and once it has gone there is nothing here left to mark with. Until
 // then both run, they agree  the server is a port of these same engines,
 // checked against them over 13,202 cases  and the server's number is the
 // one that is kept.
 //
 // Nothing about access depends on this call. /api/account-data ignores a
 // results list the client sends, so a page that skipped this, or lied
 // about the outcome, still could not record a score.
 submitAttemptForMarking(section, entry, details && details.answers);

 var updatedActiveMock = updateActiveMockSession(section, entry);
 if (updatedActiveMock) {
 abortMockSubmitFlow(section, entry);
 }

 // Dispatch a global event so the AI recommendation engine (and any other
 // listener) can react without each individual test script having to be
 // modified. Wrapped in try/catch in case CustomEvent is unavailable.
 try {
 var detail = {
 section: section,
 score: score,
 total: total,
 label: label,
 test_id: entry.test_id,
 attempt_id: entry.attempt_id,
 href: entry.href,
 submitted_at: submittedAt,
 answers: details && details.answers ? Object.assign({}, details.answers) : null,
 mock_updated: updatedActiveMock
 };
 document.dispatchEvent(new CustomEvent('ielts:test-submitted', { detail: detail }));
 } catch (e) {}
 }

 }

 function wordsFrom(value) {
 return String(value || '').trim().match(/\S+/g) || [];
 }

 function isTextAnswer(target) {
 if (!target || target.disabled || target.readOnly) return false;
 if (target.tagName === 'TEXTAREA') return true;
 if (target.tagName !== 'INPUT') return false;
 var type = String(target.type || 'text').toLowerCase();
 return type === 'text' || type === 'search';
 }

 function selectedText() {
 var active = document.activeElement;
 if (isTextAnswer(active) && typeof active.selectionStart === 'number' && typeof active.selectionEnd === 'number') {
 var inputSelection = active.value.slice(active.selectionStart, active.selectionEnd);
 if (inputSelection) return inputSelection;
 }
 return window.getSelection ? String(window.getSelection() || '') : '';
 }

 document.addEventListener('keydown', function (event) {
 var key = String(event.key || '').toLowerCase();
 if ((event.ctrlKey || event.metaKey) && !event.shiftKey && (key === 'c' || key === 'v')) {
 if (typeof event.stopImmediatePropagation === 'function') event.stopImmediatePropagation();
 event.stopPropagation();
 }
 }, true);

 document.addEventListener('copy', function (event) {
 if (typeof event.stopImmediatePropagation === 'function') event.stopImmediatePropagation();
 event.stopPropagation();
 var limited = wordsFrom(selectedText()).slice(0, 3).join(' ');
 if (!limited || !event.clipboardData) return;
 event.preventDefault();
 event.clipboardData.setData('text/plain', limited);
 }, true);

 document.addEventListener('paste', function (event) {
 if (typeof event.stopImmediatePropagation === 'function') event.stopImmediatePropagation();
 event.stopPropagation();
 var target = event.target;
 if (!isTextAnswer(target) || !event.clipboardData) return;
 event.preventDefault();

 var pasted = wordsFrom(event.clipboardData.getData('text/plain')).slice(0, 3).join(' ');
 if (!pasted) return;
 var start = typeof target.selectionStart === 'number' ? target.selectionStart : target.value.length;
 var end = typeof target.selectionEnd === 'number' ? target.selectionEnd : start;
 var combined = target.value.slice(0, start) + pasted + target.value.slice(end);
 target.value = wordsFrom(combined).slice(0, 3).join(' ');
 if (typeof target.setSelectionRange === 'function') {
 target.setSelectionRange(target.value.length, target.value.length);
 }
 target.dispatchEvent(new Event('input', { bubbles: true }));
 }, true);
 }

 // Read-only taps on the two streams: an analyser each, never connected to
 // the speakers, so nothing here changes what either person hears. One
 // AudioContext for the page, made on the first call - by then the person
 // has pressed "Find a partner", which is what lets it start.
 // The wave is the voice's spectrum wrapped round the avatar: BANDS bands,
 // log-spaced from BAND_LOW to BAND_HIGH Hz - where speech lives - down the
 // right-hand side from the top, mirrored on the left. A low hum pushes the
 // top out, an "s" the bottom, a vowel most of the ring.
 var BANDS = 32;
 var BAND_LOW = 90;
 var BAND_HIGH = 5000;

 function bandRanges(analyser) {
 var nyquist = analyser.context.sampleRate / 2;
 var bins = analyser.frequencyBinCount;
 var ranges = [];
 for (var b = 0; b < BANDS; b += 1) {
 var from = BAND_LOW * Math.pow(BAND_HIGH / BAND_LOW, b / BANDS);
 var to = BAND_LOW * Math.pow(BAND_HIGH / BAND_LOW, (b + 1) / BANDS);
 var lo = Math.max(1, Math.floor((from / nyquist) * bins));
 var hi = Math.max(lo, Math.min(bins - 1, Math.ceil((to / nyquist) * bins)));
 ranges.push([lo, hi]);
 }
 return ranges;
 }

 function watchLevel(stream, seat) {
 var Context = window.AudioContext || window.webkitAudioContext;
 if (!Context || !stream || !seat) return;
 if (meters.some(function (meter) { return meter.seat === seat; })) return;
 try {
 if (!meterContext) meterContext = new Context();
 if (meterContext.state === 'suspended') meterContext.resume();
 var source = meterContext.createMediaStreamSource(stream);
 var analyser = meterContext.createAnalyser();
 analyser.fftSize = 1024;
 // A wide decibel window, so a quiet voice across the room still
 // registers; the per-band noise floor below keeps a fan or a street
 // from moving the ring on its own.
 analyser.minDecibels = -100;
 analyser.maxDecibels = -30;
 analyser.smoothingTimeConstant = 0.55;
 source.connect(analyser);
 var floor = new Float32Array(BANDS);
 floor.fill(1);
 meters.push({
 seat: seat,
 source: source,
 analyser: analyser,
 spectrum: new Uint8Array(analyser.frequencyBinCount),
 ranges: bandRanges(analyser),
 floor: floor,
 above: new Float32Array(BANDS),
 peak: 0,
 noise: 1,
 bands: new Float32Array(BANDS),
 energy: 0,
 canvas: seat.querySelector('.partner-wave'),
 color: window.getComputedStyle(seat).getPropertyValue('--seat').trim() || '#cf3f3f',
 loudAt: 0
 });
 if (!frame) frame = window.requestAnimationFrame(tick);
 } catch (error) {
 // No wave, and nothing else lost: the call does not depend on it.
 }
 }

 // One band's reading, as how far it stands above that band's own noise.
 // The floor falls at once to anything quieter and creeps up only slowly,
 // so steady background sound becomes the floor within a few seconds and a
 // voice - which comes and goes - always stands out from it.
 function readBands(meter) {
 meter.analyser.getByteFrequencyData(meter.spectrum);
 var loudest = 0;
 for (var b = 0; b < BANDS; b += 1) {
 var range = meter.ranges[b];
 var sum = 0;
 for (var k = range[0]; k <= range[1]; k += 1) sum += meter.spectrum[k];
 var raw = sum / (range[1] - range[0] + 1) / 255;

 var floor = meter.floor[b];
 meter.floor[b] = floor + (raw - floor) * (raw < floor ? 0.3 : 0.004);
 meter.above[b] = Math.max(0, raw - meter.floor[b] - 0.02);
 if (meter.above[b] > loudest) loudest = meter.above[b];
 }

 // The gate: what the room's own noise reaches when nobody speaks, tracked
 // like the floors (down at once, up slowly), and a voice has to stand
 // clear of it. Without it the gain below amplified that noise into a
 // ring that never stopped moving. Measured on synthetic speech: silence
 // peaked at 0.06-0.09 above the floors, a quiet voice at ~0.23, a loud
 // one at ~0.5 - so the gate opens between about 0.1 and 0.16.
 meter.noise = meter.noise + (loudest - meter.noise) * (loudest < meter.noise ? 0.2 : 0.003);
 var gateFrom = meter.noise * 1.6 + 0.02;
 var open = Math.max(0, Math.min(1, (loudest - gateFrom) / 0.06));

 // Automatic gain. The loudest band sets the scale, so a whisper and a
 // shout both draw their whole shape - a fixed scale made a loud voice a
 // plain full circle and a quiet one almost nothing. How far out the ring
 // reaches is loudness, on its own curve: a quiet voice still reaches a
 // good part of the way.
 meter.peak = Math.max(open > 0 ? loudest : 0, meter.peak * 0.985);
 var reference = Math.max(meter.peak, 0.06);
 var reach = (0.3 + 0.7 * Math.min(1, Math.pow(meter.peak / 0.6, 0.8))) * open;

 var total = 0;
 for (var c = 0; c < BANDS; c += 1) {
 var value = Math.pow(Math.min(1, meter.above[c] / reference), 0.9) * reach;
 var shown = meter.bands[c];
 meter.bands[c] = shown + (value - shown) * (value > shown ? 0.55 : 0.12);
 total += meter.bands[c];
 }
 meter.energy = total / BANDS;
 }

 // One animation loop for both seats.
 function tick(time) {
 frame = 0;
 var now = Date.now();
 meters.forEach(function (meter) {
 readBands(meter);
 if (meter.energy > 0.045) meter.loudAt = now;
 meter.seat.classList.toggle('is-speaking', now - meter.loudAt < SPEAKING_HOLD_MS);
 if (!stillMotion) drawWave(meter, time);
 });
 if (meters.length) frame = window.requestAnimationFrame(tick);
 }

 // A smooth closed curve through the points: each segment a quadratic from
 // one midpoint to the next with the point itself as the control - no
 // corners, however jagged the spectrum.
 function smoothShape(context, points) {
 var count = points.length;
 var last = points[count - 1];
 context.beginPath();
 context.moveTo((last[0] + points[0][0]) / 2, (last[1] + points[0][1]) / 2);
 for (var i = 0; i < count; i += 1) {
 var point = points[i];
 var next = points[(i + 1) % count];
 context.quadraticCurveTo(point[0], point[1], (point[0] + next[0]) / 2, (point[1] + next[1]) / 2);
 }
 context.closePath();
 }

 function ringPoints(meter, centre, base, room, scale, time) {
 var points = [];
 var total = BANDS * 2;
 for (var i = 0; i < total; i += 1) {
 var band = i < BANDS ? i : total - 1 - i;
 // A little movement of its own, so a held vowel breathes rather than
 // freezing into a still shape.
 var shimmer = 0.92 + 0.08 * Math.sin(time * 0.005 + i * 0.7);
 var radius = base + room * meter.bands[band] * scale * shimmer;
 var angle = -Math.PI / 2 + (i + 0.5) / total * Math.PI * 2;
 points.push([centre + Math.cos(angle) * radius, centre + Math.sin(angle) * radius]);
 }
 return points;
 }

 // The avatar covers the middle 58% of the canvas (partner.css,
 // .partner-halo); the wave lives in the ring outside it.
 function drawWave(meter, time) {
 var canvas = meter.canvas;
 var size = canvas ? canvas.clientWidth : 0;
 if (!size) return;
 var ratio = window.devicePixelRatio || 1;
 var pixels = Math.round(size * ratio);
 if (canvas.width !== pixels) {
 canvas.width = pixels;
 canvas.height = pixels;
 }
 var context = canvas.getContext('2d');
 context.setTransform(ratio, 0, 0, ratio, 0, 0);
 context.clearRect(0, 0, size, size);

 var centre = size / 2;
 var base = size * 0.29 + 6;
 var room = centre - base - 6;
 var energy = Math.min(1, meter.energy * 3);

 // Resting: a thin ring, so the seat looks alive before anyone speaks.
 context.globalAlpha = 0.25 + 0.35 * energy;
 context.strokeStyle = meter.color;
 context.lineWidth = 1.5;
 context.beginPath();
 context.arc(centre, centre, base, 0, Math.PI * 2);
 context.stroke();
 if (meter.energy < 0.004) {
 context.globalAlpha = 1;
 return;
 }

 var outer = ringPoints(meter, centre, base, room, 1, time);
 var middle = ringPoints(meter, centre, base, room, 0.6, time + 400);

 context.fillStyle = meter.color;
 context.shadowColor = meter.color;
 context.shadowBlur = 8 + 22 * energy;
 context.globalAlpha = 0.16 + 0.1 * energy;
 smoothShape(context, outer);
 context.fill();

 context.shadowBlur = 0;
 context.globalAlpha = 0.3 + 0.15 * energy;
 smoothShape(context, middle);
 context.fill();

 // The edge, crisp: this is the line the eye reads as "the voice".
 context.globalAlpha = 0.85;
 context.lineWidth = 2;
 context.lineJoin = 'round';
 context.strokeStyle = meter.color;
 smoothShape(context, outer);
 context.stroke();
 context.globalAlpha = 1;
 }

 // --- the call ------------------------------------------------------------

 function startCall(message) {
 stopAwayTimer();
 partnerName = message.nickname;
 // A rejoin carries the partner's photo token again; one without (an
 // older server) keeps the picture already on screen instead of clearing it.
 if (message.photo || !message.rejoined) showPartnerPhoto(message.photo);
 // A fresh page coming back from a reload still has the old peer's pieces
 // in nobody's hands; the side that stayed dropped its peer on
 // partner-away. Either way this starts from a clean connection.
 if (message.rejoined) dropPeer();

 if (!message.turn) {
 // Not a degraded call: with iceTransportPolicy 'relay' and no relay
 // there is no path at all, and saying so now beats forty seconds of
 // ICE and a failure that reads like a firewall.
 setState('idle', 'The call server is not configured. Nothing can connect.');
 return;
 }
 if (!navigator.mediaDevices || !window.RTCPeerConnection) {
 setState('idle', 'This browser cannot make calls.');
 send({ type: 'hangup' });
 return;
 }

 setState('connecting', 'Found ' + partnerName + '. Asking for your microphone\u2026');

 watchdog = window.setTimeout(function () {
 watchdog = null;
 if (state !== 'connecting') return;
 var name = partnerName;
 teardown();
 setState('idle', 'Could not connect to ' + (name || 'your partner')
 + '. Press the button to look again.');
 send({ type: 'hangup' });
 }, CONNECT_MS);

 navigator.mediaDevices.getUserMedia({ audio: true, video: false })
 .then(function (stream) {
 // The call may have ended while the permission question was open -
 // the partner left, or the watchdog above gave up. A microphone
 // opened now would stay on with nothing to hang it up.
 if (state !== 'connecting') {
 stream.getTracks().forEach(function (track) { track.stop(); });
 return;
 }
 microphone = stream;
 setState('connecting', 'Connecting to ' + partnerName + '\u2026');
 if (seatMeState) seatMeState.textContent = 'Microphone on';
 watchLevel(stream, seatMe);

 peer = new RTCPeerConnection({
 iceServers: [{
 urls: message.turn.urls,
 username: message.turn.username,
 credential: message.turn.credential
 }],
 // Section 1, decision 11. Without this the browsers find each other
 // directly whenever they can, each sees the other's address, and
 // the relay is never exercised - so the setting would look fine and
 // be doing nothing.
 iceTransportPolicy: 'relay'
 });

 stream.getTracks().forEach(function (track) {
 peer.addTrack(track, stream);
 });

 peer.ontrack = function (event) {
 if (!remote) return;
 // streams[0] is there when the sender named a stream, which this
 // file always does; the fallback is for a browser that drops it.
 remote.srcObject = event.streams[0] || new MediaStream([event.track]);
 play();
 watchLevel(remote.srcObject, seatThem);
 };

 peer.onicecandidate = function (event) {
 if (event.candidate) {
 send({ type: 'ice', payload: event.candidate.toJSON() });
 }
 };

 peer.onconnectionstatechange = function () {
 if (!peer) return;
 if (peer.connectionState === 'connected') {
 if (watchdog !== null) window.clearTimeout(watchdog);
 watchdog = null;
 setState('incall', 'You are talking to ' + partnerName + '.');
 rememberCall();
 if (health === null) health = window.setInterval(checkHealth, HEALTH_MS);
 holdScreen();
 // The server starts the call on this. Both sides send it and the
 // first to arrive counts, so a partner who withholds theirs
 // cannot keep the call off the record.
 send({ type: 'connected' });
 } else if (peer.connectionState === 'failed') {
 // Waiting for the server to give the call back: the partner
 // dropped their side on our absence, and a new peer is coming
 // with the rejoin. A hangup now would end the call it holds.
 if (rejoining) {
 dropPeer();
 return;
 }
 setState('idle', 'The call could not connect. Please try again.');
 teardown();
 send({ type: 'hangup' });
 }
 };

 if (message.initiator) {
 peer.createOffer()
 .then(function (offer) { return peer.setLocalDescription(offer); })
 .then(function () {
 send({ type: 'offer', payload: peer.localDescription.sdp });
 })
 .catch(function () {
 setState('idle', 'Could not start the call.');
 teardown();
 send({ type: 'hangup' });
 });
 }

 // Whatever the partner sent while the microphone question was open,
 // in the order it came: the offer first, then its candidates.
 var waiting = early;
 early = [];
 waiting.forEach(signal);
 })
 .catch(function () {
 // Refused, or no microphone at all. The partner is released rather
 // than left waiting for an answer that is never coming.
 teardown();
 setState('idle', 'Without a microphone there is no call. '
 + 'Allow it in your browser and try again.');
 send({ type: 'hangup' });
 });
 }

 // Offer, answer or candidate from the partner. Before this side has a peer
 // they wait in `early`; startCall hands them back here once it has one.
 function signal(message) {
 if (!peer) {
 if (state === 'connecting' && partnerName) early.push(message);
 return;
 }
 if (message.type === 'offer') onOffer(message.payload);
 else if (message.type === 'answer') onAnswer(message.payload);
 else if (message.type === 'ice') onIce(message.payload);
 }

 function flushIce() {
 var waiting = pendingIce;
 pendingIce = [];
 waiting.forEach(onIce);
 }

 function onOffer(sdp) {
 if (!peer) return;
 peer.setRemoteDescription({ type: 'offer', sdp: sdp })
 .then(flushIce)
 .then(function () { return peer.createAnswer(); })
 .then(function (answer) { return peer.setLocalDescription(answer); })
 .then(function () {
 send({ type: 'answer', payload: peer.localDescription.sdp });
 })
 .catch(function () {
 setState('idle', 'Could not answer the call.');
 teardown();
 send({ type: 'hangup' });
 });
 }

 function onAnswer(sdp) {
 if (!peer) return;
 peer.setRemoteDescription({ type: 'answer', sdp: sdp }).then(flushIce).catch(function () {
 setState('idle', 'Could not finish connecting.');
 teardown();
 send({ type: 'hangup' });
 });
 }

 function onIce(candidate) {
 if (!peer) return;
 if (!peer.remoteDescription) {
 pendingIce.push(candidate);
 return;
 }
 peer.addIceCandidate(candidate).catch(function () { /* a stale one */ });
 }

 function keyBytes(base64url) {

 var bytes = new Uint8Array(raw.length);

 return bytes;

 // A socket the server closed without a word - a deploy restarting it, a

 function onMessage(event) {
 var message;
 try {
 message = JSON.parse(event.data);
 } catch (error) {
 return;
 }

 if (message.type === 'ready') {
 reconnects = 0;
 // Back after the socket dropped mid-call. The screen, the sound and
 // the clock are all still here; only the server's side is asked for.
 if (partnerName) {
 askForCall();
 return;
 }
 setState('idle', standing(message));
 if (typeof message.waiting === 'number') drawPulse(message.waiting);
 startPulse();
 // Reloaded in the middle of a call.
 if (wasInCall()) {
 setState('connecting', 'Reconnecting to your call&');
 askForCall();
 } else if (resumeLooking) {
 resumeLooking = false;
 send({ type: 'find' });
 }
 } else if (message.type === 'lobby') {
 if (state === 'idle') drawPulse(message.waiting);
 } else if (message.type === 'searching') {
 clearPulse();
 setState('looking', waitingText(message.waiting));
 if (heartbeat === null) {
 heartbeat = window.setInterval(function () {
 send({ type: 'heartbeat' });
 }, HEARTBEAT_MS);
 }
 } else if (message.type === 'cancelled') {
 stopHeartbeat();
 setState('idle', 'Not looking. Press the button when you are ready.');
 send({ type: 'lobby' });
 } else if (message.type === 'matched') {
 rejoining = false;
 stopHeartbeat();
 // A new call closes whatever the last one left open. Rating somebody
 // while talking to somebody else is the confusion this avoids.
 if (!message.rejoined) closeAfter();
 // The call's id from the pairing on, not from the connection: which
 // browser reports `connected` must not decide whether this page can
 // report or block afterwards. A rejoin carries none and keeps the one
 // this page already has (or gets it in the `call-started` after).
 if (message.call) {
 lastCall = message.call;
 lastName = message.nickname || '';
 }
 startCall(message);
 } else if (message.type === 'partner-away') {
 partnerAway(message.seconds);
 } else if (message.type === 'offer' || message.type === 'answer'
 || message.type === 'ice') {
 signal(message);
 } else if (message.type === 'call-started') {
 lastCall = message.call;
 lastName = partnerName;
 lastConnected = true;
 } else if (message.type === 'partner-left') {
 var name = partnerName;
 teardown();
 setState('idle', (name ? name + ' left.' : 'Your partner left.')
 + ' Press the button to look again.');
 openAfter();
 } else if (message.type === 'ended') {
 teardown();
 setState('idle', 'Call ended. Press the button to look again.');
 openAfter();
 } else if (message.type === 'refused') {
 told = true;
 teardown();
 setState('refused', WORDING[message.reason] || 'Not available right now.');
 } else if (message.type === 'error') {
 setState(state, WORDING[message.reason] || 'That did not work.');
 }
 }

 function onClose() {
 stopHeartbeat();
 // A refusal already put its own words on the page and closing is what
 // follows it; the page unloading needs nothing either. Neither is
 // reopened: overwriting the answer with a reconnect would replace it
 // with the noise after it.
 if (told || state === 'refused' || leaving) {
 teardown();
 return;
 }
 if (reconnects >= RECONNECT_MS.length) {
 rejoining = false;
 teardown();
 setState('error', 'Disconnected. Reload the page to try again.');
 return;
 }
 // Not teardown: the peer carries the sound without this socket, and
 // closing it here is what used to end every call on every deploy.
 if (partnerName) {
 rejoining = true;
 setState(state, 'Lost the connection to the server, reconnecting& '
 + 'The call goes on meanwhile.');
 } else {
 if (state === 'looking') resumeLooking = true;
 setState('connecting', 'Reconnecting&');
 }
 var delay = RECONNECT_MS[reconnects];
 reconnects += 1;
 window.setTimeout(function () {
 if (!openSocket()) onClose();
 }, delay);
 }
})();
