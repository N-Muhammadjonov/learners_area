+// Only module landing routes. Never prefetch attempts, answers, admin records,
+// evaluations or arbitrary private identifiers through this policy.
+const LANDINGS = new Set(["/dashboard", "/reading", "/listening", "/writing", "/speaking", "/grammar", "/vocabulary", "/video-lessons", "/progress"]);
+export function eligibleIntentTarget(raw: string, origin: string, current: string): string | null {
+  try {
+    const destination = new URL(raw, origin);
+    const here = new URL(current, origin);
+    if (destination.origin !== origin || destination.search || destination.hash || !LANDINGS.has(destination.pathname)) return null;
+    if (destination.pathname === here.pathname && destination.search === here.search) return null;
+    return destination.pathname;
+  } catch { return null; }
+}
+// Dispatch budget, NOT a promise that Next has completed an in-flight request.
+// No private data is stored here. Route names expire every minute.
+export class IntentPrefetchBudget {
+  private windowStart = 0;
+  private last = Number.NEGATIVE_INFINITY;
+  private targets = new Set<string>();
+  take(target: string, now: number): boolean {
+    if (now < this.windowStart || now - this.windowStart >= 60_000) {
+      this.windowStart = now; this.targets.clear(); this.last = Number.NEGATIVE_INFINITY;
+    }
+    if (this.targets.has(target) || this.targets.size >= 6 || now - this.last < 750) return false;
+    this.targets.add(target); this.last = now; return true;
+  }
+}

+import { z } from "zod";

+// Allows incomplete FORM drafts. This is NOT the authoritative save validator.
+// Unknown properties/versions remain in raw JSON rather than being discarded.
+const group = z.object({ id: z.string(), points: z.number().finite(), kind: z.enum(["concept", "entity", "number", "date"]), anyOf: z.array(z.string()).max(30), critical: z.boolean().optional() }).strict();
+const claim = z.object({ id: z.string(), label: z.string(), points: z.number().finite(), scope: z.enum(["sentence", "answer"]), groups: z.array(group).max(20), acceptedPhrases: z.array(z.string()).max(20).optional(), rejectPhrases: z.array(z.string()).max(20).optional() }).strict();
+const draft = z.object({ schemaVersion: z.literal(1), maxScore: z.literal(100), claims: z.array(claim).max(40) }).strict();
+export function decodeMeasurementDraft(raw: string): MeasurementStructureV1 | null {
+ if (raw.length > 200_000) return null;
+ if (!raw.trim()) return { schemaVersion: 1, maxScore: 100, claims: [] };
+ try {
+  const result = draft.safeParse(JSON.parse(raw));

32. ROTATION IDENTITY
======================================================================

Rotation uniqueness is based primarily on:

NOT:
